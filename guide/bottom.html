화면 하단에 script 추가

<script type="text/javascript">
function bottomSheet(value){
    var val = value;
    var obj = $('.bottomSheet');
    obj.close = $('.dim, .bottomSheet .close, .bottomSheet .sheetClose');
    obj.find('.inCont').removeClass('active');
    obj.find('.inCont.ty'+val).addClass('active');
    obj.addClass('active');

    obj.close.on('click', function(){
        obj.removeClass('active');
    });
};
</script>
